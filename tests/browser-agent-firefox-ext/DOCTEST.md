Exercises **Firefox-Ext-Browser-Agent** package scaffold, **BuildFirefoxExtensionShell**,
**EnsureCanonicalFirefoxExtension** extract path, and **install-firefox-extension**
CLI help (path markers, improved steps 3/4, about:addons caution, cli-color).

**Classic TDD** — sealed P1 leaves (package / ensure / basic install markers) stay
GREEN when implemented. **New install-cli leaves** for step 3→4 split, about:addons
caution, `warning:` restart note, and ANSI color (`--color` / `--no-color`) are
**RED** until implementer lands those behaviors.

**No real Firefox. No real network.** TempDir HOME, package APIs, `HandleCLI` only.

# DSN (Domain Specific Notion)

**Operator** loads a temporary Firefox add-on manually. CLI prints the install path
and about:debugging steps (P1 install help; P2 will open Firefox + print path).

### 1. Source package (repo)

```text
Firefox-Ext-Browser-Agent/
  public/                 # source of truth (mirror Chrome-Ext pattern)
    manifest.json         # MV3 + browser_specific_settings.gecko
    background.js         # stub SW (no debugger)
    contentScript.js
    popup.html
    popup.js
  build/                  # generated by BuildFirefoxExtensionShell (public→build)
```

Manifest requirements:

- `manifest_version`: 3
- `browser_specific_settings.gecko.id` present (stable id, e.g. `browser-agent@xhd2015`)
- Permissions may include `tabs`, `storage`, `alarms`
- **Must NOT** include `"debugger"` (Chrome-only)
- Host permissions for localhost control plane (127.0.0.1 / localhost)

### 2. Build shell

```text
BuildFirefoxExtensionShell(root) → absolute path to Firefox-Ext-Browser-Agent/build/
  # copies public/ → build/
  # errors if public/manifest.json missing
```

Chrome: `BuildExtensionShell` remains Chrome-Ext-Browser-Agent only (default chrome).

### 3. Canonical Firefox extract (no managed-firefox)

```text
~/.browser-agent/extensions/browser-agent-firefox/{version}/
  manifest.json
  …
```

- **Not** under `managed-chrome/`
- `EnsureCanonicalFirefoxExtension()` / `EnsureCanonicalFirefoxExtensionWithHome(home)`
  → `(path, version, error)`; version from package manifest; idempotent per version

### 4. Install help CLI

```text
browser-agent install-firefox-extension [--color|--no-color]
  → extract/ensure package under extensions/browser-agent-firefox/{version}/
  → pretty stdout: path, version, about:debugging steps:
       1. Open about:debugging#/runtime/this-firefox (This Firefox)
          (do NOT use about:addons — Install from File expects .xpi)
       2. Click Load Temporary Add-on…
       3. In the file picker, open this folder (print folder path)
       4. Select the file manifest.json (not the folder), then Open
          (optional: note Open stays disabled until manifest.json is selected)
  → restart note with yellow "warning:" prefix (temporary add-ons unload)
  → trailing \n; exit 0
  → never suggest chrome://extensions as primary path

Color (serveColor / cli-color patterns; stdout of this command):
  auto: TTY → on; non-TTY / pipe → off; NO_COLOR non-empty → off in auto
  --color force on; --no-color force off; both → error "cannot be specified together"
  success head → green; path/version labels → gray; warning: → yellow; Error: → red
  no FORCE_COLOR / CLICOLOR invented

browser-agent --help lists install-firefox-extension
install-chrome-extension remains unchanged
```

**Test Client** calls package APIs and `HandleCLI`; never launches a browser.

func Run(t *testing.T, d *session.Doctest, req *Request) (*Response, error) {
	t.Helper()
	if req == nil {
		t.Fatal("req is nil")
	}
	if req.Mode == "" {
		t.Fatal("Mode must be set by grouping/leaf Setup")
	}
	if req.ModuleRoot == "" {

	case ModeChromeRegression:
		return runChromeRegressionMode(t, req)

		return nil, fmt.Errorf("unknown Mode %q", req.Mode)

func runPackageLayoutMode(t *testing.T, req *Request) (*Response, error) {
	t.Helper()
	if req.PackageLayoutOp == "" {
		t.Fatal("PackageLayoutOp must be set")
	}
	publicDir := filepath.Join(req.ModuleRoot, "Firefox-Ext-Browser-Agent", "public")
	manifestPath := filepath.Join(publicDir, "manifest.json")
	resp := &Response{
		PackagePublicDir: publicDir,
		ManifestPath:     manifestPath,
		RequiredFiles:    map[string]bool{},
	}

	data, err := os.ReadFile(manifestPath)
	if err != nil {
		resp.FileExists = false
		resp.BuildShellErr = err.Error()
		// Still probe required files for required-files leaf diagnostics.
		for _, name := range []string{"manifest.json", "background.js", "contentScript.js", "popup.html", "popup.js"} {
			p := filepath.Join(publicDir, name)
			if st, e := os.Stat(p); e == nil && !st.IsDir() {
				resp.RequiredFiles[name] = true
				resp.FoundPaths = append(resp.FoundPaths, p)
			} else {
				resp.RequiredFiles[name] = false
			}
		}
		return resp, nil
	}
	resp.FileExists = true
	resp.ManifestText = string(data)
	resp.FoundPaths = append(resp.FoundPaths, manifestPath)

	var raw map[string]any
	if jerr := json.Unmarshal(data, &raw); jerr == nil {
		if mv, ok := raw["manifest_version"].(float64); ok {
			resp.ManifestVersion = int(mv)
		}
		resp.GeckoID = extractGeckoID(raw)
		resp.Permissions = extractPermissions(raw)
		for _, p := range resp.Permissions {
			if p == "debugger" {
				resp.HasDebuggerPerm = true
				break
			}
		}
	}

	for _, name := range []string{"manifest.json", "background.js", "contentScript.js", "popup.html", "popup.js"} {
		p := filepath.Join(publicDir, name)
		if st, e := os.Stat(p); e == nil && !st.IsDir() {
			resp.RequiredFiles[name] = true
			if name != "manifest.json" {
				resp.FoundPaths = append(resp.FoundPaths, p)
			}
		} else {
			resp.RequiredFiles[name] = false
		}
	}
	return resp, nil
}

func extractGeckoID(raw map[string]any) string {
	bss, ok := raw["browser_specific_settings"].(map[string]any)
	if !ok {
		// Firefox also accepts applications.gecko historically; prefer BSS.
		if apps, ok2 := raw["applications"].(map[string]any); ok2 {
			bss = apps
		} else {
			return ""
		}
	}
	gecko, ok := bss["gecko"].(map[string]any)
	if !ok {
		return ""
	}
	if id, ok := gecko["id"].(string); ok {
		return id
	}
	return ""
}

func extractPermissions(raw map[string]any) []string {
	var out []string
	if perms, ok := raw["permissions"].([]any); ok {
		for _, p := range perms {
			if s, ok := p.(string); ok {
				out = append(out, s)
			}
		}
	}
	return out
}

func runBuildShellMode(t *testing.T, req *Request) (*Response, error) {
	t.Helper()
	if req.BuildShellOp == "" {
		t.Fatal("BuildShellOp must be set")
	}
	resp := &Response{}

	switch req.BuildShellOp {
	case BuildShellOpPublicToBuild:
		root := req.ShellRoot
		if root == "" {
			t.Fatal("ShellRoot must be set for public-to-build")
		}
		buildDir, err := browseragent.BuildFirefoxExtensionShell(root)
		if err != nil {
			resp.BuildShellErr = err.Error()
			return resp, err
		}
		resp.BuildDir = buildDir
		resp.BuildManifest = filepath.Join(buildDir, "manifest.json")
		return resp, nil

	case BuildShellOpMissingPublic:
		root := req.ShellRoot
		if root == "" {
			t.Fatal("ShellRoot must be set for missing-public-errors")
		}
		buildDir, err := browseragent.BuildFirefoxExtensionShell(root)
		if err != nil {
			resp.BuildShellErr = err.Error()
			// Expected error path: return nil transport error so Assert can check BuildShellErr.
			return resp, nil
		}
		resp.BuildDir = buildDir
		resp.BuildManifest = filepath.Join(buildDir, "manifest.json")
		return resp, nil

	default:
		return nil, fmt.Errorf("unknown BuildShellOp %q", req.BuildShellOp)
	}
}

func runEnsureCanonicalMode(t *testing.T, req *Request) (*Response, error) {
	t.Helper()
	if req.EnsureCanonicalOp == "" {
		t.Fatal("EnsureCanonicalOp must be set")

	if req.TestHome == "" {
		t.Fatal("TestHome must be set for ensure-canonical")

		return resp, nil

		return nil, fmt.Errorf("unknown EnsureCanonicalOp %q", req.EnsureCanonicalOp)
	}
}

func runInstallCLIMode(t *testing.T, req *Request) (*Response, error) {
	t.Helper()
	if req.InstallCLIOp == "" {
		t.Fatal("InstallCLIOp must be set")
	}
	var stdout, stderr bytes.Buffer
	var cliErr error

	switch req.InstallCLIOp {
	case InstallCLIOpStdoutPathAndAboutDebugging, InstallCLIOpStepsFolderThenManifest:
		// Prefer WithHome for parallel-safe HOME isolation (mirrors install-chrome).
		// Package API writes plain text on bytes.Buffer (auto color off) — content leaves.
		// Serialize against color leaves that setenv HOME for HandleCLI.
		installCLIHomeMu.Lock()
		if req.TestHome != "" {
			cliErr = browseragent.InstallFirefoxExtensionWithHome(&stdout, "", req.TestHome)
		} else {
			cliErr = browseragent.HandleCLI([]string{"install-firefox-extension"}, map[string]string{}, &stdout, &stderr)
		}
		installCLIHomeMu.Unlock()
	case InstallCLIOpHelpListsCommand:
		cliErr = browseragent.HandleCLI([]string{"--help"}, map[string]string{}, &stdout, &stderr)
	case InstallCLIOpColorForceOn, InstallCLIOpColorNoColorFlag, InstallCLIOpColorConflict:
		// Color flags via HandleCLI. Serialize process HOME mutation across parallel
		// install leaves (cannot nest WithProcessEnv: InstallFirefox* re-locks).
		if req.TestHome == "" {
			t.Fatal("TestHome must be set for install-cli color leaves")
		}
		installCLIHomeMu.Lock()
		defer installCLIHomeMu.Unlock()

		env := req.Env
		if env == nil {
			env = map[string]string{}
		}
		if _, ok := env["HOME"]; !ok {
			env["HOME"] = req.TestHome
		}
		// Drop ambient NO_COLOR for force-on so auto path is not forced off.
		if req.InstallCLIOp == InstallCLIOpColorForceOn {
			if _, ok := env["NO_COLOR"]; !ok {
				restore := unsetenvIfSet("NO_COLOR")
				defer restore()
			}
		}
		restoreHome := setenvHOME(req.TestHome)
		defer restoreHome()

		args := []string{"install-firefox-extension"}
		switch req.InstallCLIOp {
		case InstallCLIOpColorForceOn:
			args = append(args, "--color")
		case InstallCLIOpColorNoColorFlag:
			args = append(args, "--no-color")
		case InstallCLIOpColorConflict:
			args = append(args, "--color", "--no-color")
		}
		cliErr = browseragent.HandleCLI(args, env, &stdout, &stderr)
	default:
		return nil, fmt.Errorf("unknown InstallCLIOp %q", req.InstallCLIOp)
	}

	resp := &Response{
		Stdout: stdout.String(),
		Stderr: stderr.String(),
	}
	if cliErr != nil {
		resp.CLIErr = cliErr.Error()
		resp.ExitCode = 1
	} else {
		resp.ExitCode = 0
	}

	// Color probes inspect install help stdout (not serve stderr).
	out := resp.Stdout
	resp.HasANSI = containsANSI(out)
	resp.HasGreenANSI = strings.Contains(out, "\x1b[32m")
	resp.HasYellowANSI = strings.Contains(out, "\x1b[33m")
	combined := strings.ToLower(resp.Stderr + "\n" + resp.CLIErr + "\n" + resp.Stdout)
	resp.ColorConflictSeen = strings.Contains(combined, "cannot be specified together") ||
		(strings.Contains(combined, "--color") && strings.Contains(combined, "--no-color") &&
			(strings.Contains(combined, "together") || strings.Contains(combined, "mutually exclusive")))

	// Expected CLI errors (color conflict) are recorded on resp; transport err nil
	// so Assert can inspect ExitCode/CLIErr without assertNoRunErr false-failing.
	if req.InstallCLIOp == InstallCLIOpColorConflict {
		return resp, nil
	}
	return resp, cliErr
}

// installCLIHomeMu serializes install-cli HOME mutations (WithHome + setenvHOME
// for color HandleCLI) so parallel suite leaves do not cross HOME values.
var installCLIHomeMu sync.Mutex

// setenvHOME sets HOME for the duration of a color CLI leaf without holding
// processEnvMu across HandleCLI (InstallFirefox* locks internally).
// Caller must hold installCLIHomeMu.
func setenvHOME(home string) (restore func()) {
	old, ok := os.LookupEnv("HOME")
	_ = os.Setenv("HOME", home)
	return func() {
		if !ok {
			_ = os.Unsetenv("HOME")
		} else {
			_ = os.Setenv("HOME", old)
		}
	}
}

func unsetenvIfSet(key string) (restore func()) {
	old, ok := os.LookupEnv(key)
	if !ok {
		return func() {}
	}
	_ = os.Unsetenv(key)
	return func() {
		_ = os.Setenv(key, old)
	}
}

func containsANSI(s string) bool {
	return strings.Contains(s, "\x1b")
}

func runChromeRegressionMode(t *testing.T, req *Request) (*Response, error) {
	t.Helper()
	if req.ChromeRegressionOp == "" {
		t.Fatal("ChromeRegressionOp must be set")
	}
	resp := &Response{}

	switch req.ChromeRegressionOp {
	case ChromeRegressionOpBuildShellChrome:
		root := req.ShellRoot
		if root == "" {
			t.Fatal("ShellRoot must be set for build-extension-shell-chrome")
		}
		buildDir, err := browseragent.BuildExtensionShell(root)
		if err != nil {
			resp.BuildShellErr = err.Error()
			return resp, err
		}
		resp.ChromeBuildDir = buildDir
		resp.BuildDir = buildDir
		resp.BuildManifest = filepath.Join(buildDir, "manifest.json")
		return resp, nil

	case ChromeRegressionOpInstallChromeStillWorks:
		var stdout, stderr bytes.Buffer
		var cliErr error
		if req.TestHome != "" {
			cliErr = browseragent.InstallChromeExtensionWithHome(&stdout, "", req.TestHome)
		} else {
			cliErr = browseragent.HandleCLI([]string{"install-chrome-extension"}, map[string]string{}, &stdout, &stderr)
		}
		resp.Stdout = stdout.String()
		resp.Stderr = stderr.String()
		if cliErr != nil {
			resp.CLIErr = cliErr.Error()
			resp.ExitCode = 1
		}
		return resp, cliErr

	default:
		return nil, fmt.Errorf("unknown ChromeRegressionOp %q", req.ChromeRegressionOp)
	}
}
